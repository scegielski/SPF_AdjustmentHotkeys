cmake_minimum_required(VERSION 3.16)
project(SeatDiagnostics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(SPF_SeatDiagnostics SHARED SeatDiagnostics.cpp)
target_include_directories(SPF_SeatDiagnostics PRIVATE ../../SPF_API)
target_link_libraries(SPF_SeatDiagnostics PRIVATE bcrypt)

target_include_directories(SeatDiagnosticsTests PRIVATE ../../SPF_API)
target_link_libraries(SeatDiagnosticsTests PRIVATE bcrypt)
add_test(NAME ObserverRegression COMMAND SeatDiagnosticsTests)
